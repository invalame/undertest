export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const songId = url.searchParams.get("songId");
    let id = url.searchParams.get("id"); // Legacy support
    
    const level = parseInt(url.searchParams.get("level") || "1");
    const startRaw = url.searchParams.get("start");
    const mode = url.searchParams.get("mode") || "normal";
    // ==================== CORS ===================
    const allowedOrigin = "https://undertest.vercel.app";
    if (request.method === "OPTIONS") {
      return new Response(null, {
        headers: {
          "Access-Control-Allow-Origin": allowedOrigin,
          "Access-Control-Allow-Methods": "GET, OPTIONS",
          "Access-Control-Allow-Headers": "*",
          "Access-Control-Max-Age": "86400",
        }
      });
    }
    const CORS_HEADERS = {
      "Access-Control-Allow-Origin": allowedOrigin,
      "Cache-Control": "no-store, no-cache",
    };
    // =============================================
    let objectPath = null;
    // 1. Intentar resolver por songId usando el mapping
    if (songId && typeof songMapping !== 'undefined' && songMapping[songId]) {
      objectPath = songMapping[songId].path;
    } 
    // 2. Fallback: Intentar resolver por id (Base64 legacy)
    else if (id) {
      try { id = decodeURIComponent(atob(id)); } catch (e) {}
      
      let folder = "music";
      if (mode === "album") folder = "albums";
      else if (mode === "artist") folder = "artist";
      objectPath = `${folder}/${id}`;
    }
    if (!objectPath) {
      return new Response("ID o songId no válido", { status: 400, headers: CORS_HEADERS });
    }
    const LEVEL_SECONDS = { 1: 0.5, 2: 2, 3: 4, 4: 8, 5: 11, 6: 15 };
    const seconds = LEVEL_SECONDS[level] ?? 0.5;
    const BYTES_PER_SEC = 40000;
    const bytesToFetch = Math.ceil(BYTES_PER_SEC * seconds);
    try {
      let head = await env.MUSIC_BUCKET.head(objectPath);
      // Si no se encuentra y es un ID legacy, intentar buscar en otras carpetas
      if (!head && !songId) {
        const otherFolders = ["music", "albums", "artist"].filter(f => !objectPath.startsWith(f));
        const fileName = objectPath.split('/').pop();
        for (const fbFolder of otherFolders) {
          head = await env.MUSIC_BUCKET.head(`${fbFolder}/${fileName}`);
          if (head) {
            objectPath = `${fbFolder}/${fileName}`;
            break;
          }
        }
      }
      if (!head) {
        return new Response("Archivo no encontrado", { status: 404, headers: CORS_HEADERS });
      }
      let start;
      if (!startRaw) {
        const maxStart = Math.max(0, head.size - Math.ceil(BYTES_PER_SEC * 16));
        start = Math.floor(Math.random() * maxStart);
      } else {
        start = parseInt(startRaw);
      }
      const object = await env.MUSIC_BUCKET.get(objectPath, {
        range: { offset: start, length: bytesToFetch }
      });
      return new Response(object.body, {
        headers: {
          ...CORS_HEADERS,
          "Content-Type": "audio/mpeg",
          "Access-Control-Expose-Headers": "X-Start-Byte, X-Level",
          "X-Start-Byte": String(start),
          "X-Level": String(level),
          "Content-Disposition": 'inline; filename="track.mp3"'
        }
      });
    } catch (e) {
      return new Response("Error interno", { status: 500, headers: CORS_HEADERS });
    }
  }
};
